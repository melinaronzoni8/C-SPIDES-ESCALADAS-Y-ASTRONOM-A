HTML, CSS & JS - Acordeón Animado 
<!-- HTML -->
<div class="acc-item" onclick="toggleAccordion(this)">
  <div class="acc-header">
    <span>Noche 1</span>
    <i class="fa-solid fa-chevron-down acc-icon"></i>
  </div>
  <div class="acc-content">
    <p>Contenido dinámico...</p>
  </div>
</div>

<!-- JS -->
function toggleAccordion(element) {
  const item = element.parentElement;
  const wasActive = item.classList.contains('active');
  
  // Cerrar todos los abiertos
  document.querySelectorAll('.acc-item').forEach(el => el.classList.remove('active'));
  
  if (!wasActive) {
    item.classList.add('active');
  }
}
